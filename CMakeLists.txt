function(skiffllm_apply_toolchain target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
        if(SKIFFLLM_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
        if(SKIFFLLM_ENABLE_HARDENING)
            target_compile_options(${target} PRIVATE /guard:cf /GS /sdl)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(SKIFFLLM_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
        if(SKIFFLLM_ENABLE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        endif()
        if(SKIFFLLM_ENABLE_HARDENING)
            target_compile_options(${target} PRIVATE -fstack-protector-strong)
        endif()
        if(SKIFFLLM_ENABLE_PIE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
            target_compile_options(${target} PRIVATE -fPIE)
        endif()
    endif()
endfunction()

function(skiffllm_apply_link_options target)
    if(MSVC)
        if(SKIFFLLM_ENABLE_HARDENING)
            target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /CETCOMPAT)
        endif()
    else()
        if(SKIFFLLM_ENABLE_SANITIZERS)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
        if(SKIFFLLM_ENABLE_HARDENING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
            target_link_options(${target} PRIVATE -Wl,-z,relro,-z,now)
        endif()
        if(SKIFFLLM_ENABLE_PIE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
            target_link_options(${target} PRIVATE -pie)
        endif()
    endif()
endfunction()
